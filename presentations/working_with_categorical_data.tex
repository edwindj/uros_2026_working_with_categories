\begin{frame}
  \frametitle{Working with Categories}
  
  In official Statistics categories are used a lot:
  \begin{itemize}
    \item To delineate subpopulations, e.g. gender, regions, etc.
    \item Slice and dice: break down of statistics in finer detail
    \item To make statistics comparable by using (international) standard categories, e.g. NACE, PRODCOM
  \end{itemize}
  
  {\bf Let's see how to handle categorical variables in R!}
\end{frame}

\end{frame}
\revert

\begin{frame}
  \frametitle{Categorical variable}
  \begin{itemize}
    \item Can take only a fixed number of possible values.
    \begin{description}
      \item[Educational level] Low, Middle, High
      \item[Employment status] Student, Working, Unemployed, Pensioned
      \item[Diabetes status] `Has diabetes, `Does not have diabetes'
    \end{description}
    \item In computer science (and some programming languages) called an \emph{enum} or 
      \textit{enumeration type}. 
    \item The possible values are called \emph{levels}.
    \item Important for official statistics as these define the
      (sub)populations for which the statistics are calculated.
    \item Often the possible values are from a \emph{code list}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Code list}
  \begin{itemize}
    \item Lists the possible \emph{levels} of a \emph{categorical} variable
    \item Each level has a \emph{code} and usually an descriptive \emph{label}
    \item Sometimes also description, hierarchy
    \item Datasets often use the \emph{codes}.
      \begin{itemize}
        \item More compact.
        \item Labels change more often and might be language dependent.
      \end{itemize}
    \item \emph{Labels} are used for presentation.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{\texttt{forcats}}
  
  \begin{itemize}
    \item part of (core of) \texttt{tidyverse}
    \item first argument is vector: optimized for pipe operator \texttt{|>} or \texttt{\%>\%}.
    \item all functions start with \texttt{fct_}
    \item provides (simplified) versions of common operations, e.g. \texttt{fct_inorder}, \texttt{fct_lump}.
  \end{itemize}
  
\end{frame}

\begin{frame}[fragile]{Base R vs. \texttt{forcats}: Creation \& Reordering}
  \centering
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{@{} l p{5.4cm} l @{}}
    \toprule
    \textbf{Operation} & \textbf{Base R Approach} & \textbf{\texttt{forcats} Equivalent} \\
    \midrule
    Create / Coerce & \rcode{factor(x)} / \rcode{as.factor(x)}     & \rcode{as\_factor(x)} / \rcode{fct(x)} \\
    Inspect Levels  & \rcode{levels(f)} / \rcode{table(f)}         & \rcode{fct\_count(f)} / \rcode{fct\_unique(f)} \\
    Manual Reorder  & \rcode{factor(f, levels = c("a", "b", ...))} & \rcode{fct\_relevel(f, "a", "b")} \\
    Move to Front   & \rcode{relevel(f, ref = "a")}                & \rcode{fct\_relevel(f, "a")} \\
    Order by Frequency  & \rcode{factor(f, levels = names(sort(table(f), decreasing = TRUE)))} & \rcode{fct\_infreq(f)} \\
    Order by Variable   & \rcode{reorder(f, x, FUN = median)}                                  & \rcode{fct\_reorder(f, x, .fun = median)} \\
    Order by Appearance & \rcode{factor(f, levels = unique(f))}                                & \rcode{fct\_inorder(f)} \\
    Reverse Order       & \rcode{factor(f, levels = rev(levels(f)))}                           & \rcode{fct\_rev(f)} \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}
  \frametitle{codelist}
  \begin{itemize}
    \item \texttt{codelist} objects for code lists
    \item \texttt{code} objects for vectors with code list
    \item Has the following features:
    \begin{itemize}
      \item Codes + labels + descriptions
      \item Hierarchies
      \item Codes representing missing values
      \item Multilingual labels and descriptions
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{codelist}
  \begin{itemize}
    \item Vector stores codes
    \item Functions for `casting' codes to labels 
    \item Functions for `casting' codes to codes higher in the hierarchy
    \item Comparison and assignment operators
  \end{itemize}
\end{frame}
