\begin{frame}
  \frametitle{Categorical variable}
  \begin{itemize}
    \item Can take only a fixed number of possible values.
    \begin{description}
      \item[Educational level] Low, Middle, High
      \item[Employment status] Student, Working, Unemployed, Pensioned
      \item[Diabetes status] `Has diabetes, `Does not have diabetes'
    \end{description}
    \item In computer science (and some programming languages) called an \emph{enum} or 
      \textit{enumeration type}. 
    \item The possible values are called \emph{levels}.
    \item Important for official statistics as these define the
      (sub)populations for which the statistics are calculated.
    \item Often the possible values are from a \emph{code list}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Code list}
  \begin{itemize}
    \item Lists the possible \emph{levels} of a \emph{categorical} variable
    \item Each level has a \emph{code} and usually an descriptive \emph{label}
    \item Sometimes also description, hierarchy
    \item Datasets often use the \emph{codes}.
      \begin{itemize}
        \item More compact.
        \item Labels change more often and might be language dependent.
      \end{itemize}
    \item \emph{Labels} are used for presentation.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{codelist}
  \begin{itemize}
    \item \texttt{codelist} objects for code lists
    \item \texttt{code} objects for vectors with code list
    \item Has the following features:
    \begin{itemize}
      \item Codes + labels + descriptions
      \item Hierarchies
      \item Codes representing missing values
      \item Multilingual labels and descriptions
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{codelist}
  \begin{itemize}
    \item Vector stores codes
    \item Functions for `casting' codes to labels 
    \item Functions for `casting' codes to codes higher in the hierarchy
    \item Comparison and assignment operators
  \end{itemize}
\end{frame}
